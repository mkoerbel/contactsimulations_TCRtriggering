# Model 2: 5 TCR phosphorylations
# k values for the second order CD45 and Lck reactions
# Species: TCR, CD45, Lck, pMHC (implicit)
# IDs of the TCR stay the same throughout the reations. 

#graphics opengl_good
#graphic_delay 1

###
# Variable model parameters. Comment if using Python to run
#define pMHC_mdens 0      # mol/um^2 
#define cd45_exclusion 0    # percent of exclusion inside contact, i.e 1-cd45_exclusion inside contact
#define randomint 42
#define sim_num 1           # simulation number

###
# Model parameters
define r_c 0.22        # contact radius, um
define r_b 1.0         # simulation boundary, um

# Simulation defintions
define timestep 0.002           # from Chen et al.: diffusion limited timestep, results dont schange between 0.005 and 0.001 s 
define simtime 30               # s, from Jenkins et al.: 30-70s until triggering

# definitions for TCR
define tcr_d 0.052              # um^2/s 
define tcr_mdens 100            # TCR density, mol/um^2 

# definitions for pMHC
define pMHC_d 0.33              # um^2/s 
define k_off 0.0119             # s^-1    low: 2.59
define k_on 0.00748             # um^2/(s*mol)     low: 0.0013

# definitions for pMHC-TCR complex
define complex_d 0.0            # um^2/s diffusion of pMHC-TCR complex

# definitions for CD45
define cd45_mdens 600           # mol/um^2
define cd45_d 0.113             # um^2/s 

# definitions for Lck
define lck_mdens 200            # mol/um^2
define lck_d 0.128              # um^2/s


# display
define tcr_size r_c*0.02        # display size
define tcra_size tcr_size*2.0   # display size 
define cd45_size r_c*0.02       # display size 
define lck_size r_c*0.02        # display size 

### 

# derived defintions
define a_c 3.14*r_c^2                               # contact area, um^2
define a_b 3.14*r_b^2                               # simulation area, um^2
define tcrnum_out tcr_mdens*(a_b-a_c)               # number TCR outside the contact
define tcrnum_in tcr_mdens*a_c                      # number TCR inside the contact
define pMHCnum pMHC_mdens*a_b                       # molecules
define cd45num_in cd45_mdens*(1-cd45_exclusion/100)*a_c # molecules
define cd45num_out cd45_mdens*(a_b-a_c)             # molecules
define lcknum lck_mdens*a_b                         # molecules
define cd45_proboutin 1-cd45_exclusion/100          # probability of CD45 moving from outside to insdie the contact. See example/S7_surfaces/rafts. The ratio of transmission coefficients equals the equilibrium concentration ratio. Thus set reduced outside to inside transmission. 
define cd45_probinout 1                 
define kon_pseudofirst pMHC_mdens*k_on              # pseudo first order kon rate, s^-1

# System space and time defintions
dim 2                               # 2D system
boundaries x -r_b-0.01 r_b+0.01 p   # outermost system boundaries on x axis, um
boundaries y -r_b-0.01 r_b+0.01 p   # outermost system boundaries on y axis, um
time_start 0                        # simulation starting time, s 
time_stop simtime                   # simulation stopping time, s 
time_step timestep                  # simulation step time, s 
random_seed randomint               # randomint

# Molecular species and their properties
species tcr tcrp1 tcrb tcrp1b tcrp2 tcrp2b tcrp3 tcrp3b tcrp4 tcrp4b tcrp5 tcrp5b
species cd45 lck
species_group tcrall tcr*
species_group tcrallp tcrp*
species_group tcrbound tcrb tcrp?b
species_group cd45all cd45

difc tcrall tcr_d
difc tcrbound complex_d
difc cd45 cd45_d
difc lck lck_d

display_size tcr tcr_size 
display_size tcrb tcr_size 
display_size tcrp* tcra_size
display_size cd45 cd45_size
display_size lck lck_size
color tcr lightgreen
color tcrb lightgreen
color tcrp* olive
color tcrp5* green
color cd45all maroon
color lck blue

start_surface contactborder
    color back blue             # surface color for graphical output, faces inside contact
    color front red             # front faces outside
    thickness 1                 # surface display size
    panel sph 0 0 r_c 200       # definition of the surface panel
    
    # probability of transmitting out to in
    rate_internal cd45 fsoln bsoln cd45_proboutin
    rate_internal lck fsoln bsoln 1
    rate_internal tcr fsoln bsoln 1 
    rate_internal tcrb fsoln bsoln 1 
    rate_internal tcrp1 fsoln bsoln 1
    rate_internal tcrp1b fsoln bsoln 1
    rate_internal tcrp2 fsoln bsoln 1
    rate_internal tcrp2b fsoln bsoln 1
    rate_internal tcrp3 fsoln bsoln 1
    rate_internal tcrp3b fsoln bsoln 1
    rate_internal tcrp4 fsoln bsoln 1
    rate_internal tcrp4b fsoln bsoln 1
    rate_internal tcrp5 fsoln bsoln 1
    rate_internal tcrp5b fsoln bsoln 1

    # probability of transmitting in to out
    rate_internal lck bsoln fsoln 1
    rate_internal cd45 bsoln fsoln cd45_probinout
    rate_internal tcr bsoln fsoln 1
    rate_internal tcrb bsoln fsoln 1
    rate_internal tcrp1 bsoln fsoln 1
    rate_internal tcrp1b bsoln fsoln 1
    rate_internal tcrp2 bsoln fsoln 1
    rate_internal tcrp2b bsoln fsoln 1
    rate_internal tcrp3 bsoln fsoln 1
    rate_internal tcrp3b bsoln fsoln 1
    rate_internal tcrp4 bsoln fsoln 1
    rate_internal tcrp4b bsoln fsoln 1
    rate_internal tcrp5 bsoln fsoln 1
    rate_internal tcrp5b bsoln fsoln 1

end_surface

start_surface simborder
    color back black
    color front black   
    thickness 2
    panel sphere 0 0 r_b 200
    action both all reflect
    action all both reflect
end_surface

start_compartment insidecontact         # the are within the circle
    surface contactborder               # surface that defines the border
    point 0 0                           # a point defines inside
end_compartment

start_compartment insidesim
    surface simborder
    point 0 0
end_compartment

start_compartment outsidecontact
    compartment equal insidesim
    compartment andnot insidecontact
end_compartment

## binding
# binding (keep SN)
reaction compartment=insidecontact  binding1 tcr -> tcrb kon_pseudofirst 
reaction_serialnum                  binding1 r1
reaction compartment=insidecontact  binding2 tcrp? -> tcrp$1b kon_pseudofirst 
reaction_serialnum                  binding2 r1

# unbinding (keep SN)
reaction            unbinding1 tcrb -> tcr k_off
reaction_serialnum  unbinding1 r1
reaction            unbinding2 tcrp?b -> tcrp$1 k_off
reaction_serialnum  unbinding2 r1

###
# place molecules for initial condition, start with no bound TCR
compartment_mol tcrnum_in tcr insidecontact
compartment_mol tcrnum_out tcr outsidecontact
compartment_mol cd45num_in cd45 insidecontact
compartment_mol cd45num_out cd45 outsidecontact
